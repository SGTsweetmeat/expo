import React, { useState, useRef } from 'react';

export default function DealFinderPreview() {
  const [search, setSearch] = useState('');
  const [results, setResults] = useState([]);
  const [loading, setLoading] = useState(false);
  const [photo, setPhoto] = useState(null);
  const cameraRef = useRef(null);

  const MOCK_RESULTS = [
    { id: 1, store: 'Amazon', price: 199.99, image: 'https://via.placeholder.com/150', link: 'https://amazon.com' },
    { id: 2, store: 'Walmart', price: 189.49, image: 'https://via.placeholder.com/150', link: 'https://walmart.com' },
    { id: 3, store: 'eBay', price: 175.0, image: 'https://via.placeholder.com/150', link: 'https://ebay.com' }
  ];

  const formatUSD = (n) => `$${n.toFixed(2)}`;

  const handleSearch = () => {
    setLoading(true);
    setTimeout(() => {
      const filtered = MOCK_RESULTS.filter((r) =>
        r.store.toLowerCase().includes(search.toLowerCase())
      );
      setResults(filtered);
      setLoading(false);
    }, 300);
  };

  const handleCameraInput = (e) => {
    const file = e.target.files[0];
    if (file) {
      const url = URL.createObjectURL(file);
      setPhoto(url);
    }
  };

  return (
    <div
      style={{
        fontFamily: 'Arial, sans-serif',
        padding: '5vw',
        backgroundColor: '#F8FAFC',
        minHeight: '100vh',
        boxSizing: 'border-box'
      }}
    >
      <h1 style={{ textAlign: 'center', fontSize: '6vw', marginBottom: '3vw', color: '#111' }}>DealSnap – Better Deal Finder</h1>
      <p style={{ textAlign: 'center', color: '#333', fontSize: '4vw', marginBottom: '6vw' }}>
        Take a picture or search for the product you want and find the best price across all stores.
      </p>

      <div
        style={{
          display: 'flex',
          flexDirection: 'column',
          alignItems: 'center',
          gap: '3vw',
          marginBottom: '6vw'
        }}
      >
        <label
          style={{
            background: '#2563EB',
            color: '#fff',
            padding: '3vw',
            borderRadius: '2vw',
            textAlign: 'center',
            cursor: 'pointer',
            fontSize: '4vw',
            width: '90%'
          }}
        >
          📷 Product Photo
          <input
            type="file"
            accept="image/*"
            capture="environment"
            onChange={handleCameraInput}
            style={{ display: 'none' }}
          />
        </label>

        <input
          type="text"
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          placeholder="Search product name..."
          style={{
            padding: '3vw',
            borderRadius: '2vw',
            border: '1px solid #ccc',
            width: '90%',
            fontSize: '4vw'
          }}
        />

        <button
          onClick={handleSearch}
          style={{
            width: '90%',
            padding: '3vw',
            background: '#16A34A',
            color: '#fff',
            border: 'none',
            borderRadius: '2vw',
            fontSize: '4vw'
          }}
        >
          Search
        </button>
      </div>

      {photo && (
        <div style={{ textAlign: 'center', marginBottom: '6vw' }}>
          <img src={photo} alt="Captured" style={{ width: '90%', borderRadius: '3vw' }} />
          <p style={{ color: '#555', fontSize: '4vw' }}>Captured image preview</p>
        </div>
      )}

      {loading && <p style={{ textAlign: 'center', fontSize: '4vw' }}>Loading...</p>}

      {!loading && results.length === 0 && (
        <p style={{ textAlign: 'center', color: '#777', fontSize: '4vw' }}>
          No results yet. Try searching or snapping a product photo to compare deals.
        </p>
      )}

      <div
        style={{
          display: 'flex',
          flexDirection: 'column',
          alignItems: 'center',
          gap: '6vw'
        }}
      >
        {results.map((item) => (
          <div
            key={item.id}
            style={{
              background: '#fff',
              padding: '5vw',
              borderRadius: '3vw',
              width: '90%',
              textAlign: 'center',
              boxShadow: '0 2px 6px rgba(0,0,0,0.1)'
            }}
          >
            <img src={item.image} alt={item.store} style={{ width: '100%', borderRadius: '2vw', marginBottom: '3vw' }} />
            <h3 style={{ fontSize: '5vw' }}>{item.store}</h3>
            <p style={{ color: '#16A34A', fontWeight: 'bold', fontSize: '4.5vw' }}>{formatUSD(item.price)}</p>
            <a
              href={item.link}
              target="_blank"
              rel="noopener noreferrer"
              style={{
                display: 'inline-block',
                background: '#16A34A',
                color: '#fff',
                padding: '3vw 5vw',
                borderRadius: '2vw',
                textDecoration: 'none',
                fontSize: '4vw'
              }}
            >
              View Deal
            </a>
          </div>
        ))}
      </div>
    </div>
  );
}
